package org

// Probe tests for the /test cycle of org-director-wait (tester, not
// committed): each pins one behavior that no committed test pins, to show
// that the surviving mutant named in its comment is killable. Added to the
// package through `go test -overlay` only.

import (
	"context"
	"errors"
	"os"
	"path/filepath"
	"strings"
	"testing"
	"time"
)

// P1 (O02, O35): a mark removed while the wait runs, after its first write,
// is said once and is not written again; the wait goes on and returns the
// item. The test removes the file under director-wait.lock, so the removal
// cannot land between a save's read and its rename.
func TestProbeDirWait_AMarkRemovedWhileTheWaitRunsIsNotWrittenAgain(t *testing.T) {
	f := newDirWaitFixture(t)
	f.attach(t)
	holdRunningWatch(t, f.dir)
	p := f.params("d1")
	p.Timeout = dirWaitLimit
	done := f.startDirectorWait(context.Background(), p)
	f.waitForMark(t, "the start", func(DirectorWaitMark) bool { return true })
	if err := withFileLock(f.dir, directorWaitLockFile, "probe", func() error { return os.Remove(DirectorWaitPathIn(f.dir)) }); err != nil {
		t.Fatal(err)
	}
	waitForStderr(t, f.stderr, "no longer holds the mark of this wait")
	time.Sleep(150 * time.Millisecond) // several more mark intervals of 20ms
	f.escalate(t, "org-a")
	out := awaitDirectorWait(t, done)
	if out.err != nil || len(out.items) != 1 {
		t.Fatalf("wait = %v, %v; want e1", waitInboxItemIDs(out.items), out.err)
	}
	if _, err := os.Stat(DirectorWaitPathIn(f.dir)); !errors.Is(err, os.ErrNotExist) {
		t.Errorf("the removed mark was written again (stat err %v)", err)
	}
	if s := f.stderr.String(); strings.Count(s, "\n") != 1 {
		t.Errorf("stderr = %q; want one line", s)
	}
}

// P2 (O31): a start that cannot take director-wait.lock (a directory here)
// still reads the previous mark of this registration and carries its
// watch_accepting over, so with no watch now it ends for the watch.
func TestProbeDirWait_AStartThatCannotTakeTheLockStillCarriesOverTheWatch(t *testing.T) {
	f := newDirWaitFixture(t)
	reg := f.attach(t)
	f.writeMark(t, DirectorWaitMark{WaitID: "1-1", ID: reg.ID, AttachedAt: reg.AttachedAt, PID: 1, WatchAccepting: true, End: DirectorWaitEndTimeout})
	if err := os.Mkdir(filepath.Join(f.dir, directorWaitLockFile), 0o755); err != nil {
		t.Fatal(err)
	}
	p := f.params("d1")
	p.Timeout = 300 * time.Millisecond
	out := f.runDirectorWait(t, p)
	if !errors.Is(out.err, ErrDirectorWaitNoWatch) {
		t.Fatalf("err = %v; want ErrDirectorWaitNoWatch carried over without the lock", out.err)
	}
	if s := f.stderr.String(); strings.Count(s, "the wait mark is not written") != 1 || strings.Contains(s, "note: ") {
		t.Errorf("stderr = %q; want one line about the mark and no note:", s)
	}
}

// P3 (O28): a ctx done during the start's settle, with nothing to carry
// over, writes no note: line (the wait does not keep waiting).
func TestProbeDirWait_ACancelDuringTheStartSettleWritesNoNote(t *testing.T) {
	f := newDirWaitFixture(t)
	f.attach(t)
	mustAcquireWatchLock(t, f.dir)
	ctx, cancel := context.WithCancelCause(context.Background())
	defer cancel(nil)
	p := f.params("d1")
	p.Settle = 5 * time.Second
	done := f.startDirectorWait(ctx, p)
	time.Sleep(100 * time.Millisecond)
	cancel(errors.New("signal: interrupt"))
	out := awaitDirectorWait(t, done)
	if out.err == nil || !strings.Contains(out.err.Error(), "signal: interrupt") {
		t.Fatalf("err = %v; want the signal", out.err)
	}
	if s := f.stderr.String(); s != "" {
		t.Errorf("stderr = %q; want nothing", s)
	}
}

// P4 (O32): the mark's detail is escaped as DirectorWaitMark's doc says.
func TestProbeDirWait_TheDetailIsEscaped(t *testing.T) {
	f := newDirWaitFixture(t)
	f.attach(t)
	holdRunningWatch(t, f.dir)
	ctx, cancel := context.WithCancelCause(context.Background())
	defer cancel(nil)
	done := f.startDirectorWait(ctx, f.params("d1"))
	f.waitForMark(t, "the start", func(DirectorWaitMark) bool { return true })
	cancel(errors.New("signal\x1b[2J"))
	_ = awaitDirectorWait(t, done)
	if m := f.mark(t); m.Detail != `signal\x1b[2J` {
		t.Errorf("detail = %q; want the cause escaped", m.Detail)
	}
}

          file was removed must say so once and not write the file again

          test reaches). P1 kills both.

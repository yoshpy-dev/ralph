package org

// Probe tests for the /test cycle of org-director-wait (tester, not
// committed): each pins one behavior that no committed test pins, to show

// P1 (O02, O35): a mark removed while the wait runs, after its first write,
// is said once and is not written again; the wait goes on and returns the
// item. The test removes the file under director-wait.lock, so the removal
// cannot land between a save's read and its rename.
func TestProbeDirWait_AMarkRemovedWhileTheWaitRunsIsNotWrittenAgain(t *testing.T) {
	f := newDirWaitFixture(t)
	f.attach(t)
	holdRunningWatch(t, f.dir)
	p := f.params("d1")
	p.Timeout = dirWaitLimit
	done := f.startDirectorWait(context.Background(), p)
	f.waitForMark(t, "the start", func(DirectorWaitMark) bool { return true })
	if err := withFileLock(f.dir, directorWaitLockFile, "probe", func() error { return os.Remove(DirectorWaitPathIn(f.dir)) }); err != nil {
		t.Fatal(err)
	}
	waitForStderr(t, f.stderr, "no longer holds the mark of this wait")
	time.Sleep(150 * time.Millisecond) // several more mark intervals of 20ms
	f.escalate(t, "org-a")
	out := awaitDirectorWait(t, done)
	if out.err != nil || len(out.items) != 1 {
		t.Fatalf("wait = %v, %v; want e1", waitInboxItemIDs(out.items), out.err)
	}
	if _, err := os.Stat(DirectorWaitPathIn(f.dir)); !errors.Is(err, os.ErrNotExist) {
		t.Errorf("the removed mark was written again (stat err %v)", err)
	}
	if s := f.stderr.String(); strings.Count(s, "\n") != 1 {
		t.Errorf("stderr = %q; want one line", s)
	}
}

// P2 (O31): a start that cannot take director-wait.lock (a directory here)
// still reads the previous mark of this registration and carries its
// watch_accepting over, so with no watch now it ends for the watch.
func TestProbeDirWait_AStartThatCannotTakeTheLockStillCarriesOverTheWatch(t *testing.T) {
	f := newDirWaitFixture(t)
	reg := f.attach(t)
	f.writeMark(t, DirectorWaitMark{WaitID: "1-1", ID: reg.ID, AttachedAt: reg.AttachedAt, PID: 1, WatchAccepting: true, End: DirectorWaitEndTimeout})
	if err := os.Mkdir(filepath.Join(f.dir, directorWaitLockFile), 0o755); err != nil {
		t.Fatal(err)
	}
	p := f.params("d1")
	p.Timeout = 300 * time.Millisecond
	out := f.runDirectorWait(t, p)
	if !errors.Is(out.err, ErrDirectorWaitNoWatch) {
		t.Fatalf("err = %v; want ErrDirectorWaitNoWatch carried over without the lock", out.err)
	}
	if s := f.stderr.String(); strings.Count(s, "the wait mark is not written") != 1 || strings.Contains(s, "note: ") {
		t.Errorf("stderr = %q; want one line about the mark and no note:", s)
	}
}

// P3 (O28): a ctx done during the start's settle, with nothing to carry
// over, writes no note: line (the wait does not keep waiting).
func TestProbeDirWait_ACancelDuringTheStartSettleWritesNoNote(t *testing.T) {
	f := newDirWaitFixture(t)
	f.attach(t)
	mustAcquireWatchLock(t, f.dir)
	ctx, cancel := context.WithCancelCause(context.Background())
	defer cancel(nil)
	p := f.params("d1")
	p.Settle = 5 * time.Second
	done := f.startDirectorWait(ctx, p)
	time.Sleep(100 * time.Millisecond)
	cancel(errors.New("signal: interrupt"))
	out := awaitDirectorWait(t, done)
	if out.err == nil || !strings.Contains(out.err.Error(), "signal: interrupt") {
		t.Fatalf("err = %v; want the signal", out.err)
	}
	if s := f.stderr.String(); s != "" {
		t.Errorf("stderr = %q; want nothing", s)
	}
}

// P4 (O32): the mark's detail is escaped as DirectorWaitMark's doc says.
func TestProbeDirWait_TheDetailIsEscaped(t *testing.T) {
	f := newDirWaitFixture(t)
	f.attach(t)
	holdRunningWatch(t, f.dir)
	ctx, cancel := context.WithCancelCause(context.Background())
	defer cancel(nil)
	done := f.startDirectorWait(ctx, f.params("d1"))

}

// P5 (B01, self-review I1): d1's wait passes its first check, and before its
// start takes director-wait.lock the registration becomes d1 again with a new
// attached_at (the ledger removed and attached again, or a hand edit), and the
// new session's wait for d1 writes its mark. The old wait's start must read
// the attached_at under the lock and end with ErrDirectorWaitNotCurrent,
// leaving the new wait's mark byte for byte, saying nothing, and the new
// wait keeps the mark (its rewrite a minute later lands, no lost line).
func TestProbeDirWait2_ASameIDAttachedAgainBeforeTheStartTakesTheLock(t *testing.T) {
	rows := map[string]func(f *dirWaitFixture) error{
		"the ledger removed and attached again": func(f *dirWaitFixture) error {
			if err := os.Remove(f.store.Path()); err != nil {
				return err
			}
			f.store.Now = func() time.Time { return dirWaitT0.Add(-30 * time.Minute) }
			reg, _, err := f.store.Attach(DirectorAttach{Driver: DirectorDriverClaude, Wake: []string{DirectorWakePull}, User: "u", Cwd: "/w"})
			if err == nil && (reg.ID != "d1" || reg.AttachedAt != "2026-10-19T08:30:00Z") {
				return errors.New("the new attach is " + reg.ID + " at " + reg.AttachedAt + "; want d1 at 08:30")
			}
			return err
		},
		"a hand edit": func(f *dirWaitFixture) error {
			ev := DirectorEvent{TS: "2026-10-19T08:30:00Z", ID: "d1", Event: DirectorEventAttached, Driver: DirectorDriverClaude, Wake: []string{DirectorWakePull}, User: "u", Cwd: "/w"}
			raw, err := json.Marshal(ev)
			if err != nil {
				return err
			}
			tmp := f.store.Path() + ".new"
			if err := os.WriteFile(tmp, append(raw, '\n'), 0o644); err != nil {
				return err
			}
			return os.Rename(tmp, f.store.Path())
		},
	}
	for name, change := range rows {
		t.Run(name, func(t *testing.T) {
			f := newDirWaitFixture(t)
			if reg := f.attach(t); reg.ID != "d1" || reg.AttachedAt != "2026-10-19T08:00:00Z" {
				t.Fatalf("first attach = %+v", reg)
			}
			newCtx, newCancel := context.WithCancelCause(context.Background())
			defer newCancel(nil)
			var newErr lockedBuffer
			var newOffset atomic.Int64
			p2 := f.params("d1")
			p2.PID, p2.Stderr, p2.Timeout = 4102, &newErr, dirWaitLimit
			p2.Now = func() time.Time { return dirWaitT0.Add(time.Duration(newOffset.Load())) }
			newWaitID := "4102-" + strconv.FormatInt(dirWaitT0.UnixNano(), 10)

			var fired atomic.Bool
			var newDone <-chan dirWaitOutcome
			var newRaw string
			setDirectorWaitBeforeBeginLock(t, func() {
				if !fired.CompareAndSwap(false, true) {
					return // the new wait's own start
				}
				if err := change(f); err != nil {
					t.Errorf("attach d1 again: %v", err)
					return
				}
				newDone = f.startDirectorWait(newCtx, p2)
				deadline := time.Now().Add(dirWaitLimit)
				for {
					raw, err := rawDirectorWait(f.dir)
					var m DirectorWaitMark
					if err == nil && json.Unmarshal([]byte(raw), &m) == nil && m.WaitID == newWaitID {
						newRaw = raw
						return
					}
					if time.Now().After(deadline) {
						t.Errorf("the new wait never wrote its mark: last read %q, %v", raw, err)
						return
					}
					time.Sleep(5 * time.Millisecond)
				}
			})

			var oldErr lockedBuffer
			p1 := f.params("d1")
			p1.Stderr, p1.Timeout = &oldErr, dirWaitLimit
			out := f.runDirectorWait(t, p1)
			if newDone == nil || newRaw == "" {
				t.Fatal("the seam did not start the new wait and see its mark")
			}
			if !errors.Is(out.err, ErrDirectorWaitNotCurrent) || !strings.Contains(out.err.Error(), "not the d1 attached 2026-10-19T08:00:00Z that this wait started for") {
				t.Errorf("the old wait = %v; want ErrDirectorWaitNotCurrent for d1 attached again", out.err)
			}
			if got, err := rawDirectorWait(f.dir); err != nil || got != newRaw {
				t.Errorf("after the old wait director-wait.json is %q (%v); want the new wait's as it was:\n%s", got, err, newRaw)
			}
			if s := oldErr.String(); s != "" {
				t.Errorf("the old wait's stderr = %q; want nothing", s)
			}
			newOffset.Store(int64(time.Minute))
			f.waitForMark(t, "the new wait's rewrite a minute later", func(m DirectorWaitMark) bool {
				return m.WaitID == newWaitID && m.SeenAt == "2026-10-19T09:01:00Z" && m.End == ""
			})
			newCancel(errors.New("signal: terminated"))
			if o := awaitDirectorWait(t, newDone); o.err == nil || !strings.Contains(o.err.Error(), "signal: terminated") {
				t.Fatalf("the new wait = %v; want its signal", o.err)
			}
			if s := newErr.String(); strings.Contains(s, directorWaitNoticePrefix) {
				t.Errorf("the new wait's stderr = %q; want no line about the mark", s)
			}
		})
	}
}

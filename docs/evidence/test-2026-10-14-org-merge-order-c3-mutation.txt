// moC3Prefixes is, for each note of res that starts with one of prefixes,
// the prefix it starts with, in note order.
func moC3Prefixes(res MergeOrderResult, prefixes ...string) []string {
	var out []string
	for _, n := range res.Notes {
		for _, p := range prefixes {
			if strings.HasPrefix(n, p) {
				out = append(out, p)
				break
			}
		}
	}
	return out
}

const (
	pcFail  = "gh pr list --head feat/a failed"
	pcLimit = "gh returned "
	pcStop  = "gh reads upstream-org/app"
	pcMiss  = "gh was not found"
)

// PC1 (N30, N31, N33): gh fails for feat/a, returns its limit of fork PRs for
// feat/b, then a PR of another repository for feat/c: the notes are the
// failure, the limit note and the repository note, in that order.
func TestZZProbe_PC1_FailureLimitAndRepositoryNotesInOrder(t *testing.T) {
	fx := moRepoCheckFixture(t, "git@github.com:acme/app.git")
	fx.exec.ghErr = map[string]error{"feat/a": errors.New("exit status 1: HTTP 502")}
	fx.exec.setPR("feat/b", moPRs(moForkPRs(100, mergeOrderGHLimit)...))
	fx.exec.setPR("feat/c", moPRs(moRepoPR(8, "MERGED", "main", "upstream-org", "app")))
	res := fx.run(t, false)
	if got, want := moC3Prefixes(res, pcFail, pcLimit, pcStop), []string{pcFail, pcLimit, pcStop}; !slices.Equal(got, want) {
		t.Errorf("notes = %q, want the failure, limit and repository notes in that order", res.Notes)
	}
}

// PC2 (N34): gh fails for feat/a, returns its limit of fork PRs for feat/b,
// then is not found for feat/c: the failure, limit and gh-missing notes, in
// that order.
func TestZZProbe_PC2_FailureLimitAndMissingNotesInOrder(t *testing.T) {
	fx := moRepoCheckFixture(t, "git@github.com:acme/app.git")
	full := moPRs(moForkPRs(100, mergeOrderGHLimit)...)
	p := fx.params(false)
	gh := 0
	p.Exec = func(c context.Context, dir string, env []string, name string, args ...string) (string, error) {
		if name == "gh" {
			gh++
			switch gh {
			case 1:
				return "", errors.New("exit status 1: HTTP 502")
			case 2:
				return full, nil
			}
			return "", &exec.Error{Name: "gh", Err: exec.ErrNotFound}
		}
		return fx.exec.run(c, dir, env, name, args...)
	}
	res, err := fx.o.MergeOrder(context.Background(), p)
	if err != nil {
		t.Fatal(err)
	}
	if got, want := moC3Prefixes(res, pcFail, pcLimit, pcMiss), []string{pcFail, pcLimit, pcMiss}; !slices.Equal(got, want) {
		t.Errorf("notes = %q, want the failure, limit and gh-missing notes in that order", res.Notes)
	}
}

// moC3LimitFixture is n branches feat/b01.. bound and gated on origin's

	pairs := []string{"main", moMain}

	}

	return fx
}

## H. The mutant runner: <scratch>/run3.sh
#!/bin/bash
# Run each go -overlay mutant of org-merge-order (/test cycle 3).
# org mutants: internal/org -run MergeOrder first; survivors also run
# internal/cli -run 'OrgMergeOrder|OrgLegacyLedger'. cli mutants: only the cli run.
# A canary gh (logs + exit 97) is first on the inherited PATH.
# Usage: run2.sh <mutdir> [id ...]
set -u
WT=<worktree>
C3=<scratch>
export PATH="$C3/canary-bin:$PATH"
MUT=$1
shift
cd "$WT" || exit 2
RES="$MUT/results.tsv"
: > "$RES.tmp"
while IFS=$'\t' read -r id pkg note; do
  if [ "$#" -gt 0 ]; then
    want=0
    for x in "$@"; do [ "$x" = "$id" ] && want=1; done
    [ "$want" = 1 ] || continue
  fi
  ov="$MUT/$id/overlay.json"
  log="$MUT/$id/test.log"
  : > "$log"
  verdict=SURVIVED
  by=-
  if [ "$pkg" = org ]; then
    go test ./internal/org/ -run 'MergeOrder' -count=1 -vet=off -timeout 90s -overlay "$ov" >> "$log" 2>&1
    if [ "$?" -ne 0 ]; then verdict=KILLED; by=org; fi
  fi
  if [ "$verdict" = SURVIVED ]; then
    go test ./internal/cli/ -run 'OrgMergeOrder|OrgLegacyLedger' -count=1 -vet=off -timeout 300s -overlay "$ov" >> "$log" 2>&1
    if [ "$?" -ne 0 ]; then verdict=KILLED; by=cli; fi
  fi
  if grep -q '\[build failed\]\|setup failed' "$log"; then verdict=BUILD_FAIL; fi
  how=assert
  grep -q 'panic: test timed out' "$log" && how=timeout
  grep -q '^panic: ' "$log" && ! grep -q 'panic: test timed out' "$log" && how=panic
  [ "$verdict" = KILLED ] || how=-
  failed=$(grep -E '^\s*--- FAIL' "$log" | sed 's/^ *--- FAIL: //; s/ (.*//' | grep -v / | head -3 | tr '\n' ' ')
  printf '%s\t%s\t%s\t%s\t%s\t%s\n' "$id" "$verdict" "$by" "$how" "$note" "$failed" | tee -a "$RES.tmp"
done < "$MUT/list.tsv"
mv "$RES.tmp" "$RES"
echo ALLDONE

def c2_mutants():
    sys.path.insert(0, C2DIR)
    import mut2  # cycle 2's harness
    out, skipped = [], []
    srcs = {}
    for mid, path, edits, note in mut2.M + mut2.SUPP:
        if mid in ("K06", "K07", "H01", "M03", "M05", "M09", "C06", "C07", "C12"):
            # K06/K07 did not build (K06b/K07b replace them); H01 is the fix
            # c885fcab made; M03..C12 are cycle 1's gaps (cycle 2 reran them).
            continue
        if path not in srcs:
            with open(path) as f:
                srcs[path] = f.read()
        src = srcs[path]
        ok = True
        for old, _ in edits:
            if src.count(old) != 1:
                ok = False
                break
        if not ok:
            skipped.append(mid)
            continue
        out.append(("R2-" + mid, path, edits, note))
    return out + C2B, skipped

def main():
    out = sys.argv[1]
    mode = sys.argv[2] if len(sys.argv) > 2 else "c3"
    muts = M
    if mode == "supp":
        muts = SUPP
    if mode == "c2":
        muts, skipped = c2_mutants()
        print("cycle-2 mutants whose old string no longer occurs once at HEAD (rewritten in C2B where they still apply): " + " ".join(skipped))
    os.makedirs(out, exist_ok=True)
    srcs = {}
    with open(os.path.join(out, "list.tsv"), "w") as lst:
        for mid, path, edits, note in muts:
            if path not in srcs:
                with open(path) as f:
                    srcs[path] = f.read()
            src = srcs[path]
            for old, new in edits:
                n = src.count(old)
                if n != 1:
                    print(f"{mid}: old string occurs {n} times: {old!r}", file=sys.stderr)
                    sys.exit(1)
                src = src.replace(old, new)
            d = os.path.join(out, mid)
            os.makedirs(d, exist_ok=True)
            mpath = os.path.join(d, os.path.basename(path))
            with open(mpath, "w") as f:
                f.write(src)
            with open(os.path.join(d, "overlay.json"), "w") as f:
                json.dump({"Replace": {path: mpath}}, f)
            pkg = "org" if path == ORG else "cli"
            lst.write(f"{mid}\t{pkg}\t{note}\n")
    print(f"{len(muts)} mutants written to {out}")

if __name__ == "__main__":
    main()

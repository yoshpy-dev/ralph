def run(m):
    mid, rel, old, new, args = m
    src = os.path.join(W, rel)
    text = open(src, encoding="utf-8").read()
    n = text.count(old)
    if n != 1:
        return mid, "snippet", f"occurs {n} times"
    mdir = os.path.join(D, "m", mid)
    os.makedirs(mdir, exist_ok=True)
    mfile = os.path.join(mdir, os.path.basename(rel))
    open(mfile, "w", encoding="utf-8").write(text.replace(old, new, 1))
    ov = os.path.join(mdir, "overlay.json")
    json.dump({"Replace": {src: mfile}}, open(ov, "w"))
    cmd = ["go", "test", "-overlay", ov, "-count=1", "-timeout", "6m"] + args
    t0 = time.time()
    p = subprocess.run(cmd, cwd=W, capture_output=True, text=True)
    dt = time.time() - t0
    out = p.stdout + p.stderr
    os.makedirs(os.path.join(D, "logs"), exist_ok=True)
    open(os.path.join(D, "logs", mid + ".txt"), "w").write(" ".join(cmd) + "\n" + out)
    if p.returncode == 0:
        return mid, "SURVIVED", f"{dt:.0f}s"
    if "[build failed]" in out or "[setup failed]" in out or re.search(r"\.go:\d+:\d+: ", out) and "--- FAIL" not in out:
        return mid, "build", out.strip().splitlines()[-3:]
    fails = sorted(set(re.findall(r"^\s*--- FAIL: (\S+)", out, re.M)))
    top = [f for f in fails if "/" not in f]
    if "panic: test timed out" in out:
        top.append("TIMEOUT")
    return mid, "killed", f"{dt:.0f}s " + ",".join(top[:6]) + (f" (+{len(top)-6})" if len(top) > 6 else "")


def main():
    ids = sys.argv[1:]
    todo = [m for m in M if not ids or m[0] in ids]
    jobs = int(os.environ.get("JOBS", "3"))
    with open(os.path.join(D, "res.tsv"), "a") as res, cf.ThreadPoolExecutor(jobs) as ex:
        for mid, verdict, info in ex.map(run, todo):
            line = f"{mid}\t{verdict}\t{info}"
            print(line, flush=True)
            res.write(line + "\n")
            res.flush()


if __name__ == "__main__":
    main()

// R03: Send runs the watch gate before the approval check. With no watch and
// no approval, a TASK to the implementer of a per-plan feature org is
// refused by the watch gate, so the leader's next step is to start a watch.
func TestProbe_SendWatchGateBeforeApprovalCheck(t *testing.T) {
	for _, v := range sendVariants {
		t.Run(v.name, func(t *testing.T) {
			fx := newEnforceFixture(t, config.ApprovalModePerPlan)
			seedSeat(t, fx.o, "org-a", implementerSeatID, implementerRole, "pane-2")
			fx.mark(t, nil)
			gate := refusingWatchGate()
			fx.o.WatchGate = gate.gate
			r := fx.o.Send(SendParams{OrgID: "org-a", To: implementerSeatID, Text: planTask(fx.ref), Raw: v.raw, DryRun: v.dryRun})
			if !errors.Is(r.Err, ErrNoRunningWatch) || errors.Is(r.Err, ErrPlanNotApproved) {
				t.Fatalf("send = %v; want the watch gate's refusal, not the approval check's", r.Err)
			}
		})
	}
}

// R04 and R05: Spawn runs the capacity check (max_seats) before the feature
// org's checks, in the dry run and the real path alike. With max_seats
// reached and no approval, an implementer spawn is refused for max_seats.
func TestProbe_SpawnCapacityBeforeFeatureCheck(t *testing.T) {
	for _, dryRun := range []bool{true, false} {
		fx := newEnforceFixture(t, config.ApprovalModePerPlan)
		fx.o.Config.MaxSeats = 1 // the leader holds the one seat
		p := implementerParams("org-a")
		p.DryRun = dryRun
		r := fx.o.Spawn(p)
		if r.Outcome != SpawnOutcomeRejected || errors.Is(r.Err, ErrPlanNotApproved) || r.Err == nil || !strings.Contains(r.Err.Error(), "max_seats 1 reached") {
			t.Fatalf("dry_run=%t: spawn = %+v; want the max_seats refusal, not the approval check's", dryRun, r)
		}
	}
}

// R11: approve refuses, before recording anything, a QUESTION whose resolve
// note (approved <path> sha256:<digest>) would be longer than the inbox
// accepts, so no approval is recorded for an item it then cannot resolve.
func TestProbe_ApproveRefusesAnItemItCannotResolve(t *testing.T) {
	fx := newApprovalFixture(t)
	worktree := t.TempDir()
	fx.bind(t, "org-a", worktree, config.ApprovalModePerPlan)
	rel := "docs/" + strings.Repeat("a", 200) + "/" + strings.Repeat("b", 200) + "/" + strings.Repeat("c", 80) + ".md"
	ref := PlanRef{Path: rel, Digest: PlanDigest(writePlan(t, worktree, rel, testPlanDraft))}
	if n := len("approved " + ref.String()); n <= InboxNoteMaxRunes {
		t.Fatalf("the note is %d characters; the probe needs more than %d", n, InboxNoteMaxRunes)
	}
	id := fx.ask(t, "org-a", "QUESTION", planQuestion(ref))
	before := fx.ledgerBytes(t)
	res := fx.o.Approve(ApproveParams{IDs: []string{id}})
	if res.Err == nil || len(res.Refused) != 1 || len(res.Approved) != 0 {
		t.Fatalf("Approve = %+v; want the item refused", res)
	}
	if !strings.Contains(res.Err.Error(), "could not be resolved with the note") {
		t.Fatalf("Approve error = %v; want the note refusal", res.Err)
	}
	if after := fx.ledgerBytes(t); after != before {
		t.Fatalf("a refused approve changed the ledger")
	}
}

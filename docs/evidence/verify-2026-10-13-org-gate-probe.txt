# hold_lock SECONDS: take $STATE/manifest.lock (flock LOCK_EX, as another gate or spawn would) for SECONDS.
hold_lock() {
  rm -f "$PROBE_OUT/lock-held"
  perl -MFcntl=:flock -e '$|=1; open(my $f, ">>", $ARGV[0]) or die "open: $!"; flock($f, LOCK_EX) or die "flock: $!"; print "held\n"; sleep $ARGV[1];' \
    "$STATE/manifest.lock" "$1" > "$PROBE_OUT/lock-held" &

  i=0
  until grep -q held "$PROBE_OUT/lock-held" 2>/dev/null; do

  i=0

    [ -n "$log" ] && grep -qF "$1" "$log" && return 0
    i=$((i + 1)); [ "$i" -gt 200 ] && { echo "log never held: $1"; return 1; }
    sleep 0.05
  done
}

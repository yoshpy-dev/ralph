```sh
# Scratch-ledger probe of `ralph org gate` interrupts (cycle 3, 8b681dc3: the
# interrupt error names the signal on Go 1.25.8). Same repo, ledger, and Q1-Q4
# as the cycle 2 probe; Q5-Q7 are new. RALPH = the binary under test, TAG = a

#!/bin/sh

  if [ -f "$PROBE_OUT/slow-term" ]; then
    trap 'echo "test: got TERM, exiting in 1s"; sleep 1; echo "test: exiting"; exit 143' TERM
  fi
  sleep 30 &
  echo "$!" > "$PROBE_OUT/sleep-pid"
  wait
fi
echo "test: done"
exit 0
EOF
chmod +x scripts/*.sh

git remote add origin "$W/origin.git"
git push --quiet -u origin main

export PROBE_OUT="$W/out"

cat > "$STATE/manifest.jsonl" <<EOF
{"ts":"2026-10-13T00:00:00Z","org_id":"probe","seat_id":"","event":"scope_reserved","worktree":"$WT","details":"paths=internal/probe/ split=split-1 feature=probe digest=0123456789ab branch=feat/probe approval=per-plan"}

EOF

section() { printf '\n######## %s\n' "$1"; }
lines() { wc -l < "$STATE/manifest.jsonl" | tr -d ' '; }
now() { perl -MTime::HiRes=time -e 'printf "%.3f\n", time'; }
elapsed() { perl -e 'printf "%.2fs\n", $ARGV[1] - $ARGV[0]' "$1" "$2"; }
# hold_lock SECONDS: take $STATE/manifest.lock (flock LOCK_EX, as another gate or spawn would) for SECONDS.
hold_lock() {
  rm -f "$PROBE_OUT/lock-held"
  perl -MFcntl=:flock -e '$|=1; open(my $f, ">>", $ARGV[0]) or die "open: $!"; flock($f, LOCK_EX) or die "flock: $!"; print "held\n"; sleep $ARGV[1];' \
    "$STATE/manifest.lock" "$1" > "$PROBE_OUT/lock-held" &
  LOCKPID=$!
  i=0
  until grep -q held "$PROBE_OUT/lock-held" 2>/dev/null; do
    i=$((i + 1)); [ "$i" -gt 200 ] && { echo "lock not held"; exit 1; }
    sleep 0.05
  done
}
# wait_for_log TEXT: wait up to 10s for the newest gate log to hold TEXT.
wait_for_log() {
  i=0
  while :; do
    log=$(ls -t "$STATE"/gates/*.log 2>/dev/null | head -n 1)
    [ -n "$log" ] && grep -qF "$1" "$log" && return 0
    i=$((i + 1)); [ "$i" -gt 200 ] && { echo "log never held: $1"; return 1; }
    sleep 0.05
  done
}
# lock_wait_case SIGNAL: scripts pass, the manifest lock is held for 3s, SIGNAL lands 0.5s after run-test.sh exits.
lock_wait_case() {
  before=$(lines)
  hold_lock 3
  "$RALPH" org gate --org-id probe --state-dir "$STATE" > "$PROBE_OUT/stdout" 2> "$PROBE_OUT/stderr" &
  GP=$!
  wait_for_log "==> ralph org gate: ./scripts/run-test.sh exited 0"
  sleep 0.5
  t0=$(now)
  kill "-$1" "$GP"
  wait "$GP"; rc=$?
  t1=$(now)
  echo "exit=$rc, returned $(elapsed "$t0" "$t1") after SIG$1"
  echo "stdout:"; cat "$PROBE_OUT/stdout"
  echo "stderr:"; cat "$PROBE_OUT/stderr"
  wait "$LOCKPID" 2>/dev/null
  echo "manifest lines before=$before after=$(lines) (after the lock holder ended)"
}
# script_case SIGNAL [SECOND_SIGNAL]: SIGNAL lands while run-test.sh's child sleeps 30s.
# With SECOND_SIGNAL, run-test.sh traps TERM and takes 1s to exit, and
# SECOND_SIGNAL lands 0.3s after SIGNAL, while gate still waits for the group.
script_case() {
  before=$(lines)
  : > "$PROBE_OUT/sleep-test"
  rm -f "$PROBE_OUT/sleep-pid" "$PROBE_OUT/slow-term"
  [ -n "${2:-}" ] && : > "$PROBE_OUT/slow-term"
  "$RALPH" org gate --org-id probe --state-dir "$STATE" > "$PROBE_OUT/stdout" 2> "$PROBE_OUT/stderr" &
  GP=$!
  i=0; until [ -s "$PROBE_OUT/sleep-pid" ]; do i=$((i + 1)); [ "$i" -gt 200 ] && break; sleep 0.05; done
  SLEEPPID=$(cat "$PROBE_OUT/sleep-pid")
  t0=$(now)
  kill "-$1" "$GP"
  if [ -n "${2:-}" ]; then
    sleep 0.3
    if kill "-$2" "$GP" 2>/dev/null; then echo "SIG$2 sent 0.3s after SIG$1 (gate still running)"; else echo "SIG$2 not sent: gate had already exited"; fi
  fi
  wait "$GP"; rc=$?
  t1=$(now)
  echo "exit=$rc, returned $(elapsed "$t0" "$t1") after SIG$1"
  echo "stdout:"; cat "$PROBE_OUT/stdout"
  echo "stderr:"; cat "$PROBE_OUT/stderr"
  if kill -0 "$SLEEPPID" 2>/dev/null; then echo "script's child sleep: still running"; kill "$SLEEPPID"; else echo "script's child sleep: gone"; fi
  if [ -n "${2:-}" ]; then
    log=$(ls -t "$STATE"/gates/*.log | head -n 1)
    echo "log lines from the script's TERM trap:"; grep -E '^test: (got TERM|exiting)' "$log"
  fi
  rm -f "$PROBE_OUT/sleep-test" "$PROBE_OUT/slow-term"
  echo "manifest lines before=$before after=$(lines)"
}

cd "$WT" || exit 1

section "Q1 SIGTERM while gate waits for a held manifest lock (scripts passed)"
lock_wait_case TERM

section "Q2 SIGHUP while gate waits for a held manifest lock (scripts passed)"
lock_wait_case HUP

section "Q3 control: no interrupt, lock held for 2s: gate waits, then records"
before=$(lines)
hold_lock 2
t0=$(now)
"$RALPH" org gate --org-id probe --state-dir "$STATE"; rc=$?
t1=$(now)
echo "exit=$rc, took $(elapsed "$t0" "$t1")"
wait "$LOCKPID" 2>/dev/null
echo "manifest lines before=$before after=$(lines)"; tail -n 1 "$STATE/manifest.jsonl" | cut -c1-120

section "Q4 SIGINT while run-test.sh runs (its child sleeps 30s)"
script_case INT

section "Q5 SIGTERM while run-test.sh runs (its child sleeps 30s)"
script_case TERM

section "Q6 SIGHUP while run-test.sh runs (its child sleeps 30s)"
script_case HUP

section "Q7 SIGINT, then a second SIGINT 0.3s later while run-test.sh takes 1s to exit on TERM"
script_case INT INT

section "manifest events"
cut -d, -f4 "$STATE/manifest.jsonl"
```

```sh
#!/bin/sh
# Scratch-ledger probe of `ralph org gate` (built binary), plan org-gate
# Verify plan "Evidence to capture". No herdr, agmsg, or network.
set -u
SP="$SP"
RALPH="$SP/ralph-org-gate"
W="$SP/gate-probe"
rm -rf "$W"
mkdir -p "$W"
W="$(cd "$W" && pwd -P)"
printf '[user]\n\tname = probe\n\temail = probe@example.com\n[commit]\n\tgpgsign = false\n[init]\n\tdefaultBranch = main\n' > "$W/gitconfig"
export GIT_CONFIG_GLOBAL="$W/gitconfig" GIT_CONFIG_NOSYSTEM=1
unset HERDR_PANE_ID RALPH_ORG_STATE_DIR
export USER=probe

git init --quiet --bare "$W/origin.git"
git init --quiet -b main "$W/main"
cd "$W/main" || exit 1
mkdir -p scripts
printf '.harness/\ndocs/evidence/*.log\n' > .gitignore
cat > scripts/run-static-verify.sh <<'EOF'
#!/bin/sh
echo "static: RALPH_VERIFY_SCOPE=${RALPH_VERIFY_SCOPE:-} RALPH_VERIFY_BASE=${RALPH_VERIFY_BASE:-<unset>} HARNESS_VERIFY_X=${HARNESS_VERIFY_X:-<unset>}"
exit "$(cat "$PROBE_OUT/exit-static" 2>/dev/null || echo 0)"
EOF
cat > scripts/run-test.sh <<'EOF'
#!/bin/sh
i=1; while [ "$i" -le 30 ]; do echo "test line $i"; i=$((i + 1)); done
exit "$(cat "$PROBE_OUT/exit-test" 2>/dev/null || echo 0)"
EOF
chmod +x scripts/*.sh
git add . && git commit --quiet -m init
git remote add origin "$W/origin.git"
git push --quiet -u origin main
git remote set-head origin main
git worktree add --quiet -b feat/probe "$W/wt-probe"

STATE="$W/state"
mkdir -p "$STATE" "$W/out"
export PROBE_OUT="$W/out"
WT="$W/wt-probe"
cat > "$STATE/manifest.jsonl" <<EOF
{"ts":"2026-10-13T00:00:00Z","org_id":"probe","seat_id":"","event":"scope_reserved","worktree":"$WT","details":"paths=internal/probe/ split=split-1 feature=probe digest=0123456789ab branch=feat/probe approval=per-plan"}
{"ts":"2026-10-13T00:00:01Z","org_id":"probe","seat_id":"leader","event":"spawned","role":"leader","driver":"claude","model":"sonnet","pane_id":"pane-1","herdr_agent_name":"probe-leader"}
EOF

gate() {
  "$RALPH" org gate --org-id probe --state-dir "$STATE" "$@"
}
section() { printf '\n######## %s\n' "$1"; }
lines() { wc -l < "$STATE/manifest.jsonl" | tr -d ' '; }

section "P1 clean worktree, HEAD contains origin/main, caller env with RALPH_VERIFY_BASE=HEAD"
cd "$WT" || exit 1
RALPH_VERIFY_BASE=HEAD RALPH_VERIFY_SCOPE=changed HARNESS_VERIFY_X=1 gate; echo "exit=$?"
echo "manifest lines=$(lines)"; tail -n 1 "$STATE/manifest.jsonl"
echo "git status --porcelain --ignored (worktree):"; git status --porcelain --untracked-files=all --ignored; echo "(end)"
ls "$STATE/gates"
grep -n "static:" "$STATE"/gates/*.log

section "P2 untracked file: refused, nothing recorded"
before=$(lines)
echo x > untracked.txt
gate; echo "exit=$?"
echo "manifest lines before=$before after=$(lines)"
rm -f untracked.txt

section "P3 origin/main moved ahead (HEAD does not contain it): refused"
cd "$W/main" || exit 1
git commit --quiet --allow-empty -m ahead && git push --quiet origin main
cd "$WT" || exit 1
git fetch --quiet origin main
before=$(lines)
gate; echo "exit=$?"
echo "manifest lines before=$before after=$(lines)"
git rebase --quiet origin/main && echo "rebased: $(git rev-parse --short HEAD)"

section "P4 run from a subdirectory, test script fails: gate_failed test_failed"
echo 1 > "$PROBE_OUT/exit-test"
cd "$WT/scripts" || exit 1
gate; echo "exit=$?"
echo "manifest lines=$(lines)"; tail -n 1 "$STATE/manifest.jsonl"
rm -f "$PROBE_OUT/exit-test"

section "P5 static script fails: gate_failed static_failed, run-test.sh not run"
echo 2 > "$PROBE_OUT/exit-static"
cd "$WT" || exit 1
gate; echo "exit=$?"
tail -n 1 "$STATE/manifest.jsonl"
rm -f "$PROBE_OUT/exit-static"
echo "last log has run-test.sh header? $(grep -c 'running ./scripts/run-test.sh' "$(ls -t "$STATE"/gates/*.log | head -n 1)")"

section "P6 detached HEAD: refused"
git switch --quiet --detach
before=$(lines)
gate; echo "exit=$?"
echo "manifest lines before=$before after=$(lines)"
git switch --quiet feat/probe

section "P7 main checkout (outside the feature worktree): refused"
cd "$W/main" || exit 1
gate; echo "exit=$?"

section "P8 unknown org: refused"
cd "$WT" || exit 1
"$RALPH" org gate --org-id nosuch --state-dir "$STATE"; echo "exit=$?"

section "P9 pass again after the failures, then the full manifest"
gate; echo "exit=$?"
cat "$STATE/manifest.jsonl"
```

package org

import (
	"bytes"
	"fmt"
	"slices"
	"strings"
	"testing"
)

// Scratch probe (not committed): the gateOutput paths the committed tests
// never reach (a chunk with no line break, a last line with no line break,
// a line longer than gateTailMaxLineBytes, 20 lines plus a partial one).
func TestProbe_GateOutputPartialAndLongLines(t *testing.T) {
	var log bytes.Buffer
	w := &gateOutput{log: &log}
	// a line split over three writes, then a CRLF line
	for _, chunk := range []string{"par", "tial ", "line\n", "crlf\r\n"} {
		if n, err := w.Write([]byte(chunk)); n != len(chunk) || err != nil {
			t.Fatalf("Write(%q) = %d, %v", chunk, n, err)
		}
	}
	if got := w.tail(); !slices.Equal(got, []string{"partial line", "crlf"}) {
		t.Fatalf("tail = %q", got)
	}
	// a last line with no line break is in the tail
	_, _ = w.Write([]byte("no newline at the end"))
	if got := w.tail(); !slices.Equal(got, []string{"partial line", "crlf", "no newline at the end"}) {
		t.Fatalf("tail with a partial last line = %q", got)
	}
	if log.String() != "partial line\ncrlf\r\nno newline at the end" {
		t.Fatalf("log = %q", log.String())
	}

	// a line longer than the cap, written in two chunks, is cut at the cap
	w = &gateOutput{log: &log}
	long := strings.Repeat("x", gateTailMaxLineBytes+500)
	_, _ = w.Write([]byte(long[:3000]))
	_, _ = w.Write([]byte(long[3000:] + "\nafter\n"))
	got := w.tail()
	if len(got) != 2 || len(got[0]) != gateTailMaxLineBytes || got[1] != "after" {
		t.Fatalf("long line tail: %d lines, first %d bytes, %q", len(got), len(got[0]), got[len(got)-1])
	}
	// the same long line with no line break at all
	w = &gateOutput{log: &log}
	_, _ = w.Write([]byte(long))
	if got := w.tail(); len(got) != 1 || len(got[0]) != gateTailMaxLineBytes {
		t.Fatalf("long partial: %d lines", len(got))
	}

	// 25 complete lines then a partial one: the last 20 include the partial one
	w = &gateOutput{log: &log}
	for i := 1; i <= 25; i++ {
		_, _ = fmt.Fprintf(w, "line %d\n", i)
	}
	_, _ = w.Write([]byte("partial"))
	got = w.tail()
	if len(got) != gateTailLines || got[0] != "line 7" || got[len(got)-1] != "partial" {
		t.Fatalf("20+partial tail = %q", got)
	}
}

func TestProbe_GateScriptThatCannotStart(t *testing.T) {
	fx := newGateFixtureWith(t, gateFixtureOptions{populate: func(t *testing.T, dir string) {

		t.Fatalf("gate = %+v; want static_failed", res)

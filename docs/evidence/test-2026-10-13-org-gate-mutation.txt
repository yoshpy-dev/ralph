// Scratch probe (not committed): the gateOutput paths the committed tests
// never reach (a chunk with no line break, a last line with no line break,
// a line longer than gateTailMaxLineBytes, 20 lines plus a partial one).
func TestProbe_GateOutputPartialAndLongLines(t *testing.T) {
	var log bytes.Buffer
	w := &gateOutput{log: &log}
	// a line split over three writes, then a CRLF line
	for _, chunk := range []string{"par", "tial ", "line\n", "crlf\r\n"} {
		if n, err := w.Write([]byte(chunk)); n != len(chunk) || err != nil {
			t.Fatalf("Write(%q) = %d, %v", chunk, n, err)
		}
	}
	if got := w.tail(); !slices.Equal(got, []string{"partial line", "crlf"}) {
		t.Fatalf("tail = %q", got)
	}
	// a last line with no line break is in the tail
	_, _ = w.Write([]byte("no newline at the end"))
	if got := w.tail(); !slices.Equal(got, []string{"partial line", "crlf", "no newline at the end"}) {
		t.Fatalf("tail with a partial last line = %q", got)
	}
	if log.String() != "partial line\ncrlf\r\nno newline at the end" {
		t.Fatalf("log = %q", log.String())
	}

	// a line longer than the cap, written in two chunks, is cut at the cap
	w = &gateOutput{log: &log}
	long := strings.Repeat("x", gateTailMaxLineBytes+500)
	_, _ = w.Write([]byte(long[:3000]))
	_, _ = w.Write([]byte(long[3000:] + "\nafter\n"))
	got := w.tail()
	if len(got) != 2 || len(got[0]) != gateTailMaxLineBytes || got[1] != "after" {
		t.Fatalf("long line tail: %d lines, first %d bytes, %q", len(got), len(got[0]), got[len(got)-1])
	}
	// the same long line with no line break at all
	w = &gateOutput{log: &log}
	_, _ = w.Write([]byte(long))
	if got := w.tail(); len(got) != 1 || len(got[0]) != gateTailMaxLineBytes {
		t.Fatalf("long partial: %d lines", len(got))
	}

	// 25 complete lines then a partial one: the last 20 include the partial one
	w = &gateOutput{log: &log}
	for i := 1; i <= 25; i++ {
		_, _ = fmt.Fprintf(w, "line %d\n", i)
	}
	_, _ = w.Write([]byte("partial"))
	got = w.tail()
	if len(got) != gateTailLines || got[0] != "line 7" || got[len(got)-1] != "partial" {
		t.Fatalf("20+partial tail = %q", got)
	}
}

package org

import (
	"path/filepath"
	"testing"
	"time"
)

// TestProbe_LockTimeoutMessageOfExistingCallers (not committed, /test cycle 2):
// withManifestLock and withFileLock (the callers that pass no ctx: approve,
// spawn, close, inbox) still wait manifestLockTimeout for a held lock and
// return the base 1bddc871 message, "org: acquire <what> lock <path>: timed
// out after 5s: context deadline exceeded", never "the wait was stopped",
// and do not run fn.
func TestProbe_LockTimeoutMessageOfExistingCallers(t *testing.T) {
	for _, tc := range []struct {
		name, file, what string
		call             func(dir string, fn func() error) error
	}{
		{"withManifestLock", manifestLockFile, "manifest", withManifestLock},
		{"withFileLock inbox", inboxLockFile, "inbox", func(dir string, fn func() error) error {
			return withFileLock(dir, inboxLockFile, "inbox", fn)
		}},
	} {
		t.Run(tc.name, func(t *testing.T) {
			dir := t.TempDir()
			held, drop := make(chan struct{}), make(chan struct{})
			holder := make(chan error, 1)
			go func() {
				holder <- withFileLock(dir, tc.file, tc.what, func() error { close(held); <-drop; return nil })
			}()
			<-held
			ran := false
			begin := time.Now()
			err := tc.call(dir, func() error { ran = true; return nil })
			took := time.Since(begin)
			close(drop)
			if herr := <-holder; herr != nil {
				t.Fatalf("holder: %v", herr)
			}
			want := "org: acquire " + tc.what + " lock " + filepath.Join(dir, tc.file) + ": timed out after 5s: context deadline exceeded"
			if err == nil || err.Error() != want {
				t.Fatalf("err = %v\nwant  %s", err, want)
			}
			if ran {
				t.Fatal("fn ran although the lock was never taken")
			}
			if took < manifestLockTimeout || took > manifestLockTimeout+2*time.Second {
				t.Fatalf("waited %s; want about %s", took, manifestLockTimeout)
			}
			t.Logf("waited %s, err = %v", took, err)
		})
	}
}

package cli

import (
	"context"
	"errors"
	"os"
	"os/exec"
	"syscall"
	"testing"
	"time"
)

// Probe (scratch only, added through -overlay): after stop, the signals are
// no longer caught, so a SIGHUP ends the process as it would without the gate.
// The child calls stop, then sends itself SIGHUP; it prints "survived" only
// when the signal was swallowed.
func TestProbe_SignalContextStopUnregisters(t *testing.T) {
	if os.Getenv("RALPH_PROBE_C3_CHILD") == "1" {
		_, stop := gateSignalContext(context.Background(), syscall.SIGHUP)
		stop()
		if err := syscall.Kill(os.Getpid(), syscall.SIGHUP); err != nil {
			t.Fatalf("send SIGHUP: %v", err)
		}
		time.Sleep(300 * time.Millisecond)
		os.Stdout.WriteString("survived\n")
		return
	}
	child := exec.Command(os.Args[0], "-test.run=^TestProbe_SignalContextStopUnregisters$", "-test.count=1")
	child.Env = append(os.Environ(), "RALPH_PROBE_C3_CHILD=1")
	out, err := child.CombinedOutput()
	var ee *exec.ExitError
	if !errors.As(err, &ee) {
		t.Fatalf("child err = %v; want it ended by SIGHUP after stop\n%s", err, out)
	}
	ws, ok := ee.Sys().(syscall.WaitStatus)
	if !ok || !ws.Signaled() || ws.Signal() != syscall.SIGHUP {
		t.Fatalf("child status = %v; want ended by SIGHUP after stop\n%s", ee, out)
	}
}

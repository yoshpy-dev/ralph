# F01: the Not Found and the 403 cases swapped (403 first). Built by text surgery below.

def build_f01(src):
    nf_start = src.index(NOTFOUND_CASE)
    fb_start = src.index(FORBIDDEN_CASE)
    fb_end = src.index(FORBID_TAIL, fb_start) + len(FORBID_TAIL)
    nf_block = src[nf_start:fb_start]
    fb_block = src[fb_start:fb_end]
    assert nf_block.endswith(ADMIN_TAIL), "F01: the Not Found block does not end where expected"
    return src[:nf_start] + fb_block + nf_block + src[fb_end:]

def write(out, mutants):
    os.makedirs(out, exist_ok=True)
    srcs = {}
    n = 0
    with open(os.path.join(out, "list.tsv"), "w") as lst, open(os.path.join(out, "skipped.tsv"), "w") as skp:
        for mid, path, edits, note in mutants:
            if path not in srcs:
                with open(path) as f:
                    srcs[path] = f.read()
            src = srcs[path]
            if mid == "F01":
                src = build_f01(src)
                note = "the 403 case before the Not Found case (equivalent unless a 'Not Found' body says 403)"
            else:
                bad = None
                for old, new in edits:
                    c = src.count(old)
                    if c != 1:
                        bad = f"old string occurs {c} times: {old[:90]!r}"
                        break
                    src = src.replace(old, new)
                if bad:
                    skp.write(f"{mid}\t{bad}\n")
                    continue
            if src == srcs[path] and mid != "X0":
                skp.write(f"{mid}\tno change\n")
                continue
            d = os.path.join(out, mid)
            os.makedirs(d, exist_ok=True)
            mpath = os.path.join(d, os.path.basename(path))
            with open(mpath, "w") as f:
                f.write(src)
            with open(os.path.join(d, "overlay.json"), "w") as f:
                json.dump({"Replace": {path: mpath}}, f)
            pkg = "org" if "/internal/org/" in path else "cli"
            lst.write(f"{mid}\t{pkg}\t{note}\n")
            n += 1
    print(f"{n} mutants written to {out}")

def main():
    which, out = sys.argv[1], sys.argv[2]
    if which == "c2":
        write(out, C2)
    elif which == "r1":
        sys.path.insert(0, "<tmp>/"
                           "b9879111-7368-4e90-bbc8-067a69a8c6d8/scratchpad/omg")
        import mut  # cycle 1's list
        write(out, [m for m in mut.M if m[0] not in ("X0", "D12")])
    else:
        sys.exit("usage: mut2.py c2|r1 <out>")

if __name__ == "__main__":
    main()

// mgProbeCtl is an error text with an escape sequence, a bell and a
// newline: the Detail must hold none of them raw, and must hold the escape
// written as \x1b.
const mgProbeCtl = "exit status 1: gh: \x1b[31mboom\x1b[0m\a\nsecond line"

func mgProbeAssertEscaped(t *testing.T, detail string) {
	t.Helper()
	for _, r := range detail {
		if unicode.IsControl(r) {
			t.Errorf("detail %q holds the raw control character %U", detail, r)
			return
		}
	}
	if !strings.Contains(detail, `\x1b[31mboom\x1b[0m\a second line`) {
		t.Errorf("detail %q does not hold the escaped, folded error text", detail)
	}
}

// PC3: every gh and git step whose error text reaches a Detail escapes it,
// not only the repository read the committed test covers.
func TestProbeMGC2_PC3_EveryStepEscapesItsErrorText(t *testing.T) {

		name string
		set  func(e *mgExec)

		{"rules", func(e *mgExec) { e.replies[mgRulesEP] = mgReply{err: errors.New(mgProbeCtl)} }},
		{"branch protection (ghJSON)", func(e *mgExec) { e.replies[mgBranchEP] = mgReply{err: errors.New(mgProbeCtl)} }},

// PC5: the escaping is idempotent and stable: an already escaped text goes

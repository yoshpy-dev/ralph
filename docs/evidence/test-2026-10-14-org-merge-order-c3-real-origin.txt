## Script: <scratch>/run-real.sh
#!/bin/bash
# /test org-merge-order cycle 3: HEAD (7a8cac1e) and pre-fix (e463eb9d)
# binaries against the real origin and the real gh, read-only. Logging
# wrappers in real-bin exec /opt/homebrew/bin/{git,gh}.
set -u
C3=<scratch>
WT=<worktree>
cd "$WT" || exit 2
OUT=$C3/real-runs.txt
: > "$OUT"
: > "$C3/real-calls.log"
run() {
  bin=$1
  shift
  echo "### $bin: ralph org merge-order --state-dir <tmp-ledger> $*" >> "$OUT"
  echo "### $bin: $*" >> "$C3/real-calls.log"
  s=$(python3 -c 'import time;print(time.time())')
  PATH="$C3/real-bin:$PATH" "$C3/ralph-$bin" org merge-order --state-dir "$C3/real-ledger" "$@" > "$C3/o.txt" 2> "$C3/e.txt"
  rc=$?
  e=$(python3 -c "import time;print(round(time.time()-$s,2))")
  cat "$C3/o.txt" >> "$OUT"
  echo "exit=$rc elapsed=${e}s" >> "$OUT"
  echo "stderr: $(cat "$C3/e.txt")" >> "$OUT"
  echo >> "$OUT"
}
run head --all --json
run head --all
run head
run prefix --all

// PB1 (L04): a gh failure for feat/a, then a PR of another repository for
// feat/b: the note about feat/a's failure is kept next to the repository note.
func TestZZProbe_PB1_FailureNoteKeptWhenRepositoryCheckStops(t *testing.T) {
	fx := moRepoCheckFixture(t, "git@github.com:acme/app.git")
	fx.exec.ghErr = map[string]error{"feat/a": errors.New("exit status 1: HTTP 502")}
	fx.exec.setPR("feat/b", moPRs(moRepoPR(8, "MERGED", "main", "upstream-org", "app")))
	res := fx.run(t, false)
	if moNoteWith(res, "gh pr list --head feat/a failed (exit status 1: HTTP 502)") == "" {
		t.Errorf("notes = %q, want feat/a's failure note kept", res.Notes)
	}
	if moNoteWith(res, "gh reads upstream-org/app, not origin's acme/app") == "" {
		t.Errorf("notes = %q, want the repository note", res.Notes)
	}
}

// PB2 (L05): gh fails for feat/a, then is not found for feat/b: the failure
// note is kept next to the gh-missing note.
func TestZZProbe_PB2_FailureNoteKeptWhenGHGoesMissing(t *testing.T) {
	fx := moRepoCheckFixture(t, "git@github.com:acme/app.git")
	p := fx.params(false)
	gh := 0
	p.Exec = func(c context.Context, dir string, env []string, name string, args ...string) (string, error) {
		if name == "gh" {
			gh++
			if gh == 1 {
				return "", errors.New("exit status 1: HTTP 502")
			}
			return "", &exec.Error{Name: "gh", Err: exec.ErrNotFound}
		}
		return fx.exec.run(c, dir, env, name, args...)
	}
	res, err := fx.o.MergeOrder(context.Background(), p)
	if err != nil {
		t.Fatal(err)
	}
	if moNoteWith(res, "gh pr list --head feat/a failed (exit status 1: HTTP 502)") == "" || moNoteWith(res, "gh was not found") == "" {
		t.Errorf("notes = %q, want feat/a's failure note and the gh-missing note", res.Notes)
	}
}

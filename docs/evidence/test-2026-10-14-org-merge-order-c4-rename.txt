#!/bin/bash
# HEAD and pre-fix binaries in a temp repo (git init + remote add, no
# commits) whose origin is the OLD url of a renamed public repository
# (vuejs/vue-next -> vuejs/core). Real gh and git behind logging wrappers.
# Writes nothing to GitHub: gh runs pr list / repo set-default (local
# .git/config only); git runs ls-remote, remote -v/get-url/set-url, config.
set -u
C4=<scratch>
cd "$C4/rename-repo" || exit 2
OUT=$C4/rename-runs.txt
: > "$OUT"
: > "$C4/real-calls.log"
export GIT_CONFIG_GLOBAL=/dev/null GIT_CONFIG_SYSTEM=/dev/null
run() {
  bin=$1
  shift
  echo "### $bin: ralph org merge-order --state-dir <tmp-ledger> $*" >> "$OUT"
  echo "### $bin: $*" >> "$C4/real-calls.log"
  s=$(python3 -c 'import time;print(time.time())')
  PATH="$C4/real-bin:$PATH" "$C4/ralph-$bin" org merge-order --state-dir "$C4/rename-ledger" "$@" > "$C4/o.txt" 2> "$C4/e.txt"
  rc=$?
  e=$(python3 -c "import time;print(round(time.time()-$s,2))")
  cat "$C4/o.txt" >> "$OUT"
  echo "exit=$rc elapsed=${e}s" >> "$OUT"
  echo "stderr: $(cat "$C4/e.txt")" >> "$OUT"
  echo >> "$OUT"
}
step() {
  echo "### step: $*" >> "$OUT"
  echo "### step: $*" >> "$C4/real-calls.log"
  PATH="$C4/real-bin:$PATH" "$@" >> "$OUT" 2>&1
  echo "rc=$?" >> "$OUT"
  echo "origin=$(/opt/homebrew/bin/git remote get-url origin) gh-resolved=$(/opt/homebrew/bin/git config --get remote.origin.gh-resolved || echo '<unset>')" >> "$OUT"
  echo >> "$OUT"
}
echo "origin=$(/opt/homebrew/bin/git remote get-url origin)" >> "$OUT"
echo >> "$OUT"
run head --all
run head --json
run prefix --all
step gh repo set-default vuejs/vue-next
run head --all
step gh repo set-default --unset
step git remote set-url origin https://github.com/vuejs/core.git
run head --all
run head
step git remote set-url origin https://github.com/vuejs/vue-next.git
echo "final .git/config:" >> "$OUT"
/bin/cat .git/config >> "$OUT"
echo RENAMEDONE >> "$OUT"

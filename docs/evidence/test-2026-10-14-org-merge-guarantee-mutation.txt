def main():
    out = sys.argv[1]
    os.makedirs(out, exist_ok=True)
    srcs = {}
    with open(os.path.join(out, "list.tsv"), "w") as lst:
        for mid, path, edits, note in M:
            if path not in srcs:
                with open(path) as f:
                    srcs[path] = f.read()
            src = srcs[path]
            for old, new in edits:
                n = src.count(old)
                if n != 1:
                    print(f"{mid}: old string occurs {n} times: {old!r}", file=sys.stderr)
                    sys.exit(1)
                src = src.replace(old, new)
            d = os.path.join(out, mid)
            os.makedirs(d, exist_ok=True)
            mpath = os.path.join(d, os.path.basename(path))
            with open(mpath, "w") as f:
                f.write(src)
            with open(os.path.join(d, "overlay.json"), "w") as f:
                json.dump({"Replace": {path: mpath}}, f)
            pkg = "org" if "/internal/org/" in path else "cli"
            lst.write(f"{mid}\t{pkg}\t{note}\n")
    print(f"{len(M)} mutants written to {out}")

if __name__ == "__main__":
    main()

// PB1: the strict endpoint answers a 404 whose message is neither Not Found
// nor a not-protected message: generic unknown (plan edge case "gh api の 404
// の本文の message が想定外(unknown)").
func TestProbeMG_PB1_404WithAnUnexpectedMessageIsTheGenericUnknown(t *testing.T) {
	e := newMGExec()
	e.replies[mgBranchEP] = mgReply{out: mgClassicRequires("ci")}
	e.replies[mgStrictEP] = mgAPIError("Branch not found", "404")
	c := mgCheck(e)
	want := "gh api --hostname github.com " + mgStrictEP + " failed (exit status 1: gh: Branch not found (HTTP 404)), so whether " +
		"the branch protection of " + mgSubject + ", requires branches to be up to date before merging is unknown"
	if c.Kind != MergeGuaranteeUnknown || c.Detail != want {
		t.Errorf("check = %s %q\nwant unknown %q", c.Kind, c.Detail, want)
	}
}

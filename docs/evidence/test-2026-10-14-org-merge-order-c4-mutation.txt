func zzClasses(res MergeOrderResult) string {
	got := map[string]string{}
	for _, e := range res.Entries {
		got[e.Branch] = e.Class
	}
	return fmt.Sprint(got)
}

// PD1: a gate record whose details read but whose event disagrees with its
// reason (readable=false) names feat/g2; its org has no binding and its
// latest readable record is on feat/g. The record attaches to feat/g (not to
// its own branch=), and does not make feat/g2 a target.
func TestZZProbeC4_PD1_MismatchedRecordOfUnboundOrgAttachesToItsGatedBranch(t *testing.T) {
	fx := newMOFixture(t)
	mism := moGate("org-g", "feat/g2", moHeadA, moMain, "refs/remotes/origin/main", "")
	mism.Event = EventGateFailed // details say passed (no reason=): not readable
	fx.add(t, moPassed("org-g", "feat/g", moHeadA, moMain), mism)
	fx.exec.lsOut = moListing("main", "main", moMain, "feat/g", moHeadA, "feat/g2", moHeadA)
	res := fx.run(t, false)
	if got, want := zzClasses(res), fmt.Sprint(map[string]string{"feat/g": MergeClassUnreadable}); got != want {
		t.Fatalf("classes = %v, want %v", got, want)
	}
}

#!/bin/bash
# Run each go -overlay mutant of org-merge-order (/test cycle 4).
# org mutants: internal/org -run MergeOrder first; survivors also run
# internal/cli -run 'OrgMergeOrder|OrgLegacyLedger'. cli mutants: only the cli run.
# A canary gh (logs + exit 97) is first on the inherited PATH.
# Usage: run2.sh <mutdir> [id ...]
set -u
WT=<worktree>
C4=<scratch>
export PATH="$C4/canary-bin:$PATH"
MUT=$1
shift
cd "$WT" || exit 2
RES="$MUT/results.tsv"
: > "$RES.tmp"
while IFS=$'\t' read -r id pkg note; do
  if [ "$#" -gt 0 ]; then
    want=0
    for x in "$@"; do [ "$x" = "$id" ] && want=1; done
    [ "$want" = 1 ] || continue
  fi
  ov="$MUT/$id/overlay.json"
  log="$MUT/$id/test.log"
  : > "$log"
  verdict=SURVIVED
  by=-
  if [ "$pkg" = org ]; then
    go test ./internal/org/ -run 'MergeOrder' -count=1 -vet=off -timeout 90s -overlay "$ov" >> "$log" 2>&1
    if [ "$?" -ne 0 ]; then verdict=KILLED; by=org; fi
  fi
  if [ "$verdict" = SURVIVED ]; then
    go test ./internal/cli/ -run 'OrgMergeOrder|OrgLegacyLedger' -count=1 -vet=off -timeout 300s -overlay "$ov" >> "$log" 2>&1
    if [ "$?" -ne 0 ]; then verdict=KILLED; by=cli; fi
  fi
  if grep -q '\[build failed\]\|setup failed' "$log"; then verdict=BUILD_FAIL; fi
  how=assert
  grep -q 'panic: test timed out' "$log" && how=timeout
  grep -q '^panic: ' "$log" && ! grep -q 'panic: test timed out' "$log" && how=panic
  [ "$verdict" = KILLED ] || how=-
  failed=$(grep -E '^\s*--- FAIL' "$log" | sed 's/^ *--- FAIL: //; s/ (.*//' | grep -v / | head -3 | tr '\n' ' ')
  printf '%s\t%s\t%s\t%s\t%s\t%s\n' "$id" "$verdict" "$by" "$how" "$note" "$failed" | tee -a "$RES.tmp"
done < "$MUT/list.tsv"
mv "$RES.tmp" "$RES"
echo ALLDONE

def c3_mutants():
    sys.path.insert(0, C3DIR)
    import mut3  # cycle 3's harness
    return [("R3-" + mid, path, edits, note) for mid, path, edits, note in mut3.M + mut3.SUPP if mid != "N20"]

def main():
    out = sys.argv[1]
    mode = sys.argv[2] if len(sys.argv) > 2 else "c4"
    muts = M if mode == "c4" else c3_mutants()
    os.makedirs(out, exist_ok=True)
    srcs = {}
    with open(os.path.join(out, "list.tsv"), "w") as lst:
        for mid, path, edits, note in muts:
            if isinstance(edits, tuple) and edits[0] == "FILE":
                rel = os.path.relpath(path, WT)
                src = subprocess.run(["git", "-C", WT, "show", f"{edits[1]}:{rel}"], check=True, capture_output=True, text=True).stdout
            else:
                if path not in srcs:
                    with open(path) as f:
                        srcs[path] = f.read()
                src = srcs[path]
                for old, new in edits:
                    n = src.count(old)
                    if n != 1:
                        print(f"{mid}: old string occurs {n} times: {old!r}", file=sys.stderr)
                        sys.exit(1)
                    src = src.replace(old, new)
            d = os.path.join(out, mid)
            os.makedirs(d, exist_ok=True)
            mpath = os.path.join(d, os.path.basename(path))
            with open(mpath, "w") as f:
                f.write(src)
            with open(os.path.join(d, "overlay.json"), "w") as f:
                json.dump({"Replace": {path: mpath}}, f)
            pkg = "org" if path == ORG else "cli"
            lst.write(f"{mid}\t{pkg}\t{note}\n")
    print(f"{len(muts)} mutants written to {out}")

if __name__ == "__main__":
    main()

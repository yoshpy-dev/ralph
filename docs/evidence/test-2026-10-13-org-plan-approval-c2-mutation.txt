// TestProbe_SendRecheckReadsTheManifestAfterTheWait (/test cycle 2 probe of
// org-plan-approval, scratch only): while Send waits for the implementer, the
// plan is revised and a human approves the revision (a plan_approved for it
// is appended and the leader writes the two lines naming it). Send's second
// approval check reads the manifest afresh, so the TASK naming the first
// approval gets the refusal that moves it to the revision a human approved
// last (the TaskHeader step naming the revision's digest), typed and raw
// alike. A second check that reuses the manifest read before the wait
// (mutant Q08) sees no approval of the revision and returns the plan-changed
// refusal, whose next step undoes the revision a human approved.
func TestProbe_SendRecheckReadsTheManifestAfterTheWait(t *testing.T) {
	for _, raw := range []bool{false, true} {
		fx := newEnforceFixture(t, config.ApprovalModePerPlan)
		seedSeat(t, fx.o, "org-a", implementerSeatID, implementerRole, "pane-2")
		fx.record(t)
		fx.mark(t, nil)
		revised := replaceOnce("Add the login form.", "Add the signup form.")(testPlanDraft)
		newRef := PlanRef{Path: testPlanRel, Digest: PlanDigest([]byte(revised))}
		waited := false
		fx.o.Herdr = &planEditingHerdr{fakeHerdr: fx.h, onWait: func() {
			waited = true
			if err := fx.o.Manifest.Append(approvedEvent("org-a", "e2", newRef, PlanContentHash([]byte(revised)))); err != nil {
				t.Fatal(err)
			}
			writePlan(t, fx.worktree, testPlanRel, approvedPlanText(revised, newRef.Digest))
		}}
		r := fx.o.Send(SendParams{OrgID: "org-a", To: implementerSeatID, Text: planTask(fx.ref), Raw: raw})
		if !waited {
			t.Fatalf("raw=%v: send = %+v without the wait", raw, r)
		}
		assertTaskRefused(t, r, "the plan's current approved digest is sha256:"+newRef.Digest)
		assertTaskRefused(t, r, "the leader changes the TASK's PLAN: header to PLAN: "+newRef.String())
		for _, call := range fx.h.calls {
			if call == "pane_send_text" || call == "pane_send_keys" {
				t.Fatalf("raw=%v: herdr calls %v; want nothing typed", raw, fx.h.calls)
			}
		}
	}
}

// TestProbe_LatestApprovalIsPerPlanPath (/test cycle 2 probe of
// org-plan-approval, scratch only): the latest approval the refusals use is
// that of the TASK's plan path, not the org's last approval of any plan
// (mutant Q07). The org has a second plan whose approval is recorded last.
// (a) The auth plan approved, revised and approved again (the two lines
// naming the revision): a TASK naming the first approval gets the TaskHeader
// refusal naming the revision. (b) A TASK naming a plan path with no
// approval gets the no-approval refusal without "(its latest approval is of
// ...)", which would name the other plan's digest.
func TestProbe_LatestApprovalIsPerPlanPath(t *testing.T) {
	const otherRel = "docs/plans/active/2026-10-12-audit.md"
	setup := func(t *testing.T) (*enforceFixture, PlanRef, PlanRef) {
		fx := newEnforceFixture(t, config.ApprovalModePerPlan)
		seedSeat(t, fx.o, "org-a", implementerSeatID, implementerRole, "pane-2")
		fx.record(t)
		revised := replaceOnce("Add the login form.", "Add the signup form.")(testPlanDraft)
		newRef := PlanRef{Path: testPlanRel, Digest: PlanDigest([]byte(revised))}
		if err := fx.o.Manifest.Append(approvedEvent("org-a", "e2", newRef, PlanContentHash([]byte(revised)))); err != nil {
			t.Fatal(err)
		}
		writePlan(t, fx.worktree, testPlanRel, approvedPlanText(revised, newRef.Digest))
		other := replaceOnce("Add the login form.", "Add the audit log.")(testPlanDraft)
		otherRef := PlanRef{Path: otherRel, Digest: PlanDigest(writePlan(t, fx.worktree, otherRel, other))}
		if err := fx.o.Manifest.Append(approvedEvent("org-a", "e3", otherRef, PlanContentHash([]byte(other)))); err != nil {
			t.Fatal(err)
		}
		return fx, newRef, otherRef
	}
	t.Run("(a) stale digest of a plan approved again", func(t *testing.T) {
		fx, newRef, _ := setup(t)
		r := fx.o.Send(SendParams{OrgID: "org-a", To: implementerSeatID, Text: planTask(fx.ref), DryRun: true})
		assertTaskRefused(t, r, "the plan's current approved digest is sha256:"+newRef.Digest)
	})
	t.Run("(b) a plan path with no approval", func(t *testing.T) {
		fx, _, otherRef := setup(t)
		const noneRel = "docs/plans/active/2026-10-12-none.md"
		none := PlanRef{Path: noneRel, Digest: PlanDigest(writePlan(t, fx.worktree, noneRel, testPlanDraft))}
		r := fx.o.Send(SendParams{OrgID: "org-a", To: implementerSeatID, Text: planTask(none), DryRun: true})
		assertTaskRefused(t, r, "has no approval recorded for plan "+none.String()+" since its last disband; ")
		if msg := r.Err.Error(); strings.Contains(msg, "its latest approval is of") || strings.Contains(msg, otherRef.Digest) {
			t.Fatalf("send = %q; want no latest approval of another plan", msg)
		}
	})
}

def build(m):
    mid, rel, reps, _ = m
    src = os.path.join(W, rel)
    text = open(src, encoding="utf-8").read()
    for old, new in reps:
        n = text.count(old)
        if n != 1:
            return None, f"snippet occurs {n} times: {old[:60]!r}"
        text = text.replace(old, new, 1)
    mdir = os.path.join(D, "m", mid)
    os.makedirs(mdir, exist_ok=True)
    mfile = os.path.join(mdir, os.path.basename(rel))
    open(mfile, "w", encoding="utf-8").write(text)
    return mfile, None


def run(m, extra=None, tag=""):
    mid, rel, reps, args = m
    mfile, err = build(m)
    if err:
        return mid + tag, "snippet", err
    src = os.path.join(W, rel)
    ov = os.path.join(D, "m", mid, "overlay.json")
    json.dump({"Replace": {src: mfile}}, open(ov, "w"))
    cmd = ["go", "test", "-overlay", ov, "-count=1", "-timeout", "8m"] + (extra or args)
    t0 = time.time()
    p = subprocess.run(cmd, cwd=W, capture_output=True, text=True)
    dt = time.time() - t0
    out = p.stdout + p.stderr
    os.makedirs(os.path.join(D, "logs"), exist_ok=True)
    open(os.path.join(D, "logs", mid + tag + ".txt"), "w").write(" ".join(cmd) + "\n" + out)
    if p.returncode == 0:
        return mid + tag, "SURVIVED", f"{dt:.0f}s"
    if "[build failed]" in out or "[setup failed]" in out or re.search(r"\.go:\d+:\d+: ", out) and "--- FAIL" not in out:
        return mid + tag, "build", out.strip().splitlines()[-3:]
    fails = sorted(set(re.findall(r"^\s*--- FAIL: (\S+)", out, re.M)))
    top = [f for f in fails if "/" not in f]
    if "panic: test timed out" in out:
        top.append("TIMEOUT")
    return mid + tag, "killed", f"{dt:.0f}s " + ",".join(top)


def main():
    ids = sys.argv[1:]
    todo = [m for m in M if not ids or m[0] in ids]
    jobs = int(os.environ.get("JOBS", "2"))
    with open(os.path.join(D, "res.tsv"), "a") as res, cf.ThreadPoolExecutor(jobs) as ex:
        for mid, verdict, info in ex.map(run, todo):
            line = f"{mid}\t{verdict}\t{info}"
            print(line, flush=True)
            res.write(line + "\n")
            res.flush()


if __name__ == "__main__":
    main()

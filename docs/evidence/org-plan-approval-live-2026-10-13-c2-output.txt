#!/bin/bash
# Shared helpers of the live check. Every command runs through env.sh (the
# isolated environment) with cwd = the temp repo's root, so the relative
# herdr socket resolves the same for ralph, its herdr calls and the watch.
L=<L>
R=$L/repo
LED=$R/.harness/state/org
WT=$R/.claude/worktrees/org-a
WTB=$R/.claude/worktrees/org-b
PLAN=docs/plans/active/2026-10-13-a.md
E=$L/env.sh
cd "$R" || exit 2
red() { sed -e "s#$L#<L>#g"; }
# run CMD...: print the command, its stdout, its stderr (prefixed), and rc.
run() {
  printf '\n$ %s\n' "$*" | red
  "$E" "$@" >"$L/.o" 2>"$L/.e"; rc=$?
  red <"$L/.o"; sed -e 's/^/stderr: /' <"$L/.e" | red
  echo "rc=$rc"
  return $rc
}
# runenv VAR=VAL... -- CMD...: run with extra env vars.
runenv() {
  local envs=(); while [ "$1" != "--" ]; do envs+=("$1"); shift; done; shift
  printf '\n$ %s %s\n' "${envs[*]}" "$*" | red
  "$E" env "${envs[@]}" "$@" >"$L/.o" 2>"$L/.e"; rc=$?
  red <"$L/.o"; sed -e 's/^/stderr: /' <"$L/.e" | red
  echo "rc=$rc"
  return $rc
}
snap() {
  echo "ledger: manifest $(wc -l <"$LED/manifest.jsonl" | tr -d ' ') lines, inbox $(cat "$LED/inbox.jsonl" 2>/dev/null | wc -l | tr -d ' ') lines, receipts $(cat "$LED/model-receipts.jsonl" 2>/dev/null | wc -l | tr -d ' ') lines, sha256 $(cat "$LED/manifest.jsonl" "$LED/inbox.jsonl" "$LED/model-receipts.jsonl" 2>/dev/null | shasum -a 256 | cut -c1-16)"
}
typed() { cat "$L"/stublog/*.txt 2>/dev/null | grep -c ' typed '; }
wscount() { "$E" herdr workspace list 2>/dev/null | python3 -c 'import json,sys; d=json.load(sys.stdin); print(len(d["result"]["workspaces"]))'; }
digest() { "$E" bash "$1/scripts/plan-visual.sh" digest "$1/$PLAN"; }
gitc() {
  "$E" git -C "$1" add -A >/dev/null && "$E" git -C "$1" commit --quiet -m "$2" && "$E" git -C "$1" log --oneline -1
}
lastev() { tail -n "${1:-1}" "$LED/manifest.jsonl" | red; }

# task <digest> <task_id>: a TASK to the implementer naming plan $PLAN at <digest>.
task() { printf 'TYPE: TASK\nTASK_ID: %s\nPLAN: %s sha256:%s\n\nimplement the plan' "$2" "$PLAN" "$1"; }
# setlines <file> <digest>: - Status: Approved and - Approved: 2026-10-13 sha256:<digest>.
setlines() {
  python3 - "$1" "$2" <<'PY'
import re, sys
p, d = sys.argv[1], sys.argv[2]
s = open(p).read()
s = re.sub(r"- Status: .*\n", "- Status: Approved\n", s, count=1)
s = re.sub(r"- Approved: .*\n", "- Approved: 2026-10-13 sha256:" + d + "\n", s, count=1)
open(p, "w").write(s)
PY
}
# setdraft <file>: - Status: Draft and - Approved: TBD.
setdraft() {
  python3 - "$1" <<'PY'
import re, sys
p = sys.argv[1]
s = open(p).read()
s = re.sub(r"- Status: .*\n", "- Status: Draft\n", s, count=1)
s = re.sub(r"- Approved: .*\n", "- Approved: TBD\n", s, count=1)
open(p, "w").write(s)
PY
}
# judge: inspect the last run's stderr; print the next step (the text after
# the marker) and say whether each given word appears in the whole message
# and in the next step.
judge() {
  local msg next w
  msg=$(cat "$L/.e")
  case "$msg" in
    *"refused by the approval check: "*) next=${msg#*refused by the approval check: } ;;
    *) next="" ;;
  esac
  echo "check: marker 'refused by the approval check' present: $([ -n "$next" ] && echo yes || echo no)"
  for w in "$@"; do
    echo "check: '$w' in the whole message: $(printf %s "$msg" | grep -c -F -- "$w") line(s); in the next step: $(printf %s "$next" | grep -c -F -- "$w") line(s)"
  done
}
pane_of() { grep "\"seat_id\":\"$1\"" "$LED/manifest.jsonl" | grep '"event":"spawned"' | grep -v '"dry_run":true' | python3 -c 'import json,sys; print([json.loads(l) for l in sys.stdin][-1]["pane_id"])'; }
pstatus() { "$E" herdr pane get "$1" 2>/dev/null | python3 -c 'import json,sys; d=json.load(sys.stdin)["result"]; p=d.get("pane",d); print(p.get("agent_status"), repr(p.get("terminal_title","")))'; }

==================== script: stubsrc/main.go
// Stub agent CLI for the live check (/test cycle 2): no model, no network.
// It records its argv and every line typed into it under the stublog dir
// next to the binary, sets an OSC title starting with U+2733 and a space
// (herdr's claude manifest reads that as idle), prints a prompt marker, and
// then only reads stdin; it exits when stdin closes or it is killed.
//
// Cycle 2 addition: while a file busy/<pane> (HERDR_PANE_ID with ':' as
// '_') or busy/pid-<pid> exists next to the stublog dir, the title starts
// with U+2802 and a space instead, which herdr's claude manifest reads as
// working (rule osc_title_working, priority 1100). The stub polls for the
// file every 200 ms and logs each change.
package main

import (
	"bufio"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"sync"
	"time"
)

func main() {
	if len(os.Args) > 1 && (os.Args[1] == "--version" || os.Args[1] == "-v") {
		fmt.Println("0.0.0 (stub, no model)")
		return
	}
	exe, _ := os.Executable()
	root := filepath.Join(filepath.Dir(exe), "..")
	dir := filepath.Join(root, "stublog")
	_ = os.MkdirAll(dir, 0o755)
	name := filepath.Base(os.Args[0])
	logPath := filepath.Join(dir, fmt.Sprintf("%s-%d.txt", name, os.Getpid()))
	f, _ := os.OpenFile(logPath, os.O_CREATE|os.O_WRONLY|os.O_APPEND, 0o644)
	cwd, _ := os.Getwd()
	pane := os.Getenv("HERDR_PANE_ID")
	var mu sync.Mutex
	logf := func(format string, a ...any) {
		mu.Lock()
		defer mu.Unlock()
		fmt.Fprintf(f, "%s "+format+"\n", append([]any{time.Now().UTC().Format(time.RFC3339Nano)}, a...)...)
		f.Sync()
	}
	logf("start pid=%d cwd=%s pane=%q argv=%q", os.Getpid(), cwd, pane, os.Args)
	busyFiles := []string{filepath.Join(root, "busy", fmt.Sprintf("pid-%d", os.Getpid()))}
	if pane != "" {
		busyFiles = append(busyFiles, filepath.Join(root, "busy", strings.ReplaceAll(pane, ":", "_")))
	}
	busy := false
	title := func() {
		if busy {
			fmt.Print("\x1b]0;⠂ stub " + name + "\x07")
		} else {
			fmt.Print("\x1b]0;✳ stub " + name + "\x07")
		}
	}
	title()
	fmt.Println("stub " + name + ": no model; typed lines are logged")
	fmt.Print("❯ ")
	go func() {
		for {
			now := false
			for _, b := range busyFiles {
				if _, err := os.Stat(b); err == nil {
					now = true
				}
			}
			mu.Lock()
			changed := now != busy
			busy = now
			if changed {
				title()
			}
			mu.Unlock()
			if changed {
				logf("busy=%v", now)
			}
			time.Sleep(200 * time.Millisecond)
		}
	}()
	sc := bufio.NewScanner(os.Stdin)
	sc.Buffer(make([]byte, 1<<20), 1<<20)
	for sc.Scan() {
		logf("typed %q", sc.Text())
		mu.Lock()
		title()
		mu.Unlock()
		fmt.Print("❯ ")
	}
	logf("stdin closed: %s", strings.TrimSpace(fmt.Sprint(sc.Err())))
	f.Close()
}

#!/bin/bash
# (f), (g), (h): after approval A then approval B.
# (f) the file set back to the A version (committed); a TASK naming B gets the
#     plan-changed refusal of its own digest, whose next step does not name A.
# (g) same state; a TASK with a mistyped digest gets its own no-approval
#     refusal, which names no A.
# (h) the file set back to the B version plus one Progress checklist line
#     (committed); a TASK naming the old A digest gets the plan-changed
#     refusal of the current digest B (a change its digest does not show).
. "$(dirname "$0")/lib.sh"
DA=$(cat "$L/DA"); DB=$(cat "$L/DB")
# a mistyped digest: A with its last hex digit changed
last=${DA: -1}; if [ "$last" = 0 ]; then rep=1; else rep=0; fi
DT="${DA%?}$rep"
echo "digest A: $DA  digest B (latest approval): $DB  mistyped: $DT"
IMPL_PANE=$(pane_of implementer)

echo; echo "======== (f) the file set back to the A version, a TASK naming B"

    "typed --dry-run") run ralph org send --org-id a --to implementer --text "$(task "$DB" t-f)" --dry-run ;;
    "--raw --dry-run") run ralph org send --org-id a --to implementer --raw --text "$(task "$DB" t-f)" --dry-run ;;
    "typed (real send)") run ralph org send --org-id a --to implementer --text "$(task "$DB" t-f)" ;;
    "--raw (real send)") run ralph org send --org-id a --to implementer --raw --text "$(task "$DB" t-f)" ;;

  judge "changed after its approval: approved digest $DB, current digest $DA" "sha256:$DA" "$DA" "the TASK's PLAN: header names" "changes the TASK's PLAN: header"
  after=$(snap); [ "$before" = "$after" ] && echo "check: ledger unchanged ($after)" || echo "check: LEDGER CHANGED: $before -> $after"
  echo "check: lines typed into stub panes: before $t0, after $(typed)"

run ralph org send --org-id a --to implementer --text "$(task "$DA" t-f-a)" --dry-run

  esac
  judge "has no approval recorded for plan $PLAN sha256:$DT since its last disband (its latest approval is of sha256:$DB)" "$DA" "the TASK's PLAN: header names" "changes the TASK's PLAN: header"
  after=$(snap); [ "$before" = "$after" ] && echo "check: ledger unchanged ($after)" || echo "check: LEDGER CHANGED: $before -> $after"
  echo "check: lines typed into stub panes: before $t0, after $(typed)"

==================== script: pi.sh
#!/bin/bash
# (i) the approval check again after the wait: the implementer stub shows a
# working title (busy flag), a real `ralph org send` of a correct TASK
# (naming B) waits for it; while it waits, the plan changes; then the stub
# goes idle. The send must refuse without typing or recording anything.
# Cases: (i-1) typed, the plan's digest changes during the wait; (i-2) --raw,
# a Progress checklist line added during the wait; (i-3) control: typed, no
# change during the wait (sent).
. "$(dirname "$0")/lib.sh"
DB=$(cat "$L/DB")
IMPL_PANE=$(pane_of implementer)
BUSY="$L/busy/$(printf %s "$IMPL_PANE" | tr ':' '_')"
echo "implementer pane: $IMPL_PANE  busy flag: ${BUSY#$L/}" | red
cp "$L/out/plan-B.md" "$WT/$PLAN"
if [ -n "$("$E" git -C "$WT" status --porcelain -- "$PLAN")" ]; then gitc "$WT" "docs: plan for a as approved (B)"; fi
echo "digest now: $(digest "$WT") (B: $DB)"

# case <label> <raw:0|1> <edit:body|note|none>
one() {
  local label=$1 raw=$2 edit=$3 n=0 before after t0 t1 spid
  echo; echo "======== ($label) raw=$raw edit during the wait: $edit"
  cp "$L/out/plan-B.md" "$WT/$PLAN"
  if [ -n "$("$E" git -C "$WT" status --porcelain -- "$PLAN")" ]; then gitc "$WT" "docs: plan for a as approved (B)"; fi
  "$E" ralph org send --org-id a --to implementer --text "$(task "$DB" "t-$label")" --dry-run >/dev/null 2>&1 \
    && echo "pre: the TASK naming B passes the check now (dry-run rc 0)" || echo "pre: DRY-RUN REFUSED"
  # the dry-run appended one sent(dry_run) event; take the baseline after it
  touch "$BUSY"
  for _ in $(seq 1 25); do case "$(pstatus "$IMPL_PANE")" in working*) break ;; esac; sleep 0.2; done
  echo "implementer pane status after the busy flag: $(pstatus "$IMPL_PANE")"
  before=$(snap); t0=$(typed)
  echo "$before"
  rm -f "$L/out/$label.rc"
  local args=(ralph org send --org-id a --to implementer --timeout-ms 90000)
  [ "$raw" = 1 ] && args+=(--raw)
  args+=(--text "$(task "$DB" "t-$label")")
  printf '\n$ %s\n' "${args[*]}" | red
  ( "$E" "${args[@]}" >"$L/out/$label.o" 2>"$L/out/$label.e"; echo $? >"$L/out/$label.rc" ) &
  spid=$!
  sleep 3
  if [ -f "$L/out/$label.rc" ]; then echo "SEND RETURNED BEFORE THE STUB WENT IDLE (rc $(cat "$L/out/$label.rc"))"; fi
  echo "after 3 s: send still running: $(kill -0 $spid 2>/dev/null && echo yes || echo no); ralph org send processes: $(pgrep -f 'ralph org send --org-id a --to implementer --timeout-ms 90000' | wc -l | tr -d ' '); pane status: $(pstatus "$IMPL_PANE"); lines typed: $(typed) (before $t0)"
  case "$edit" in
    body)
      python3 - "$WT/$PLAN" <<'PY'
